```compact
pragma language_version 0.28;
import CompactStandardLibrary;

export sealed ledger owner: Bytes<32>;
export ledger cap: Uint<64>;
export ledger spent: Uint<64>;

witness secretKey(): Bytes<32>;

constructor(limit: Uint<64>) {
  assert limit > 0, "Limit must be greater than zero";
  owner = persistentHash<Vector<2, Bytes<32>>>([pad(32, "shadow-budget:owner:v1"), secretKey()]);
  cap = disclose(limit);
}

export pure circuit ownerKey(sk: Bytes<32>): Bytes<32> {
  return persistentHash<Vector<2, Bytes<32>>>([pad(32, "shadow-budget:owner:v1"), sk]);
}

export circuit spend(amount: Uint<64>): [] {
  assert amount > 0, "Amount must be greater than zero";
  const remaining = cap - spent;
  assert amount <= remaining, "Spending exceeds available budget";
  spent = disclose(spent + amount) as Uint<64>;
}
```
