export pure circuit ownerKey(sk: Bytes<32>): Bytes<32> {
  return persistentHash<Vector<2, Bytes<32>>>([pad(32, "shadow-budget:owner:v1"), sk]);
}
export circuit spend(amount: Uint<64>): [] {
  assert(ownerKey(secretKey()) == owner, "Not authorized");
  assert(amount > 0, "Amount must be positive");
  assert(amount <= cap - spent, "Insufficient funds");
  spent = (spent + disclose(amount)) as Uint<64>;
}
