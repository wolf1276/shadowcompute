import CompactStandardLibrary;

export sealed ledger cap: Uint<64>;

  assert(cap > 0, "Limit must be greater than zero");
  owner = disclose(ownerKey(secretKey()));
}
export pure circuit ownerKey(sk: Bytes<32>): Bytes<32> {
  return persistentHash<Vector<2, Bytes<32>>>([pad(32, "shadow-budget:owner:v1"), sk]);
}
export circuit spend(amount: Uint<64>): [] {
  assert(ownerKey(secretKey()) == owner, "Not owner");
  assert(amount > 0, "Amount must be greater than zero");
  assert(amount <= cap - spent, "Insufficient budget remaining");
  const delta = disclose(amount) as Uint<64>;
  spent = (spent + delta) as Uint<64>;
}
